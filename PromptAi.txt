intent("start login", async p => {
    p.play({command:'start login'})
    let usernameConfirmed = false;
//     let passwordConfirmed = false;
    
    while (!usernameConfirmed) {
        p.play("Sure! Please provide your username.");
        const username = await p.then(getEmail);
        p.play(`You provided the username: ${username}. Is that correct?`);
        const confirmation = await p.then(confirm);
        if (confirmation.value === "yes") {
            usernameConfirmed = true;
            p.play({ command: 'getUsername' ,data:username});
            
        } else {
            p.play("Okay, let's try again.");
        }
    }
