intent("join lecture",async p => {
     p.play({command:'joinMeet'}) 
     let meetCodeConfirmed = false;
    while (!meetCodeConfirmed) {
       
        p.play("Please provide the meet code.");
        const meetCode = await p.then(getMeetCode);
          console.log(meetCode)
       
        p.play(`You entered the meet code: ${meetCode}. Is that correct?`);
        const confirmation = await p.then(confirm);
       
        if (confirmation.value.toLowerCase() === "yes") {
            meetCodeConfirmed = true;
             p.play({ command: 'inMeet' ,data:meetCode});
            p.play("You are in the meet.");
           
        } else {
           
            p.play("Okay, please enter the meet code again.");
           
            meetCodeConfirmed = false;
        }
    }
});

// // Context function to capture the meet code
const getMeetCode = context(() => {
    intent("$(MEETNUMBER* .+)", p => p.resolve(p.MEETNUMBER));
});
