const setLoginStatus = (p, { loginStatus }) => {
    // Use the loginStatus parameter to update the login status in your logic
    if (loginStatus) {
        p.play("You are logged in successfully.");
    } else {
        p.play("Login failed. Please try again later.");
    }
};

// Add the method to the project API
projectAPI.setLoginStatus = setLoginStatus;
